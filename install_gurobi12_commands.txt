# 安装 MacroEnergy 环境并配置 Gurobi 12.0.0 的命令
# 在服务器上直接复制粘贴以下命令执行
# 
# 安装顺序说明：
# 1. 先加载 Gurobi 模块（构建时需要）
# 2. 安装 MacroEnergy 及其核心依赖
# 3. 然后添加和构建 Gurobi（因为 Gurobi 是可选依赖）

# ========== 第一部分：准备 Gurobi 环境 ==========
# 1. 加载模块
module purge
module load gurobi/12.0.0
module load julia/1.12.1

# 2. 设置 Gurobi 环境变量
export GUROBI_HOME=$(dirname $(dirname $(which gurobi_cl)))
export LD_LIBRARY_PATH=${GUROBI_HOME}/lib:${LD_LIBRARY_PATH}
export PATH=${GUROBI_HOME}/bin:${PATH}

# 3. 设置 OpenMP 线程数（避免 libgomp 警告）
#    如果 OMP_NUM_THREADS 未设置或无效，设置为 1
export OMP_NUM_THREADS=${OMP_NUM_THREADS:-1}

# 4. 验证 Gurobi 版本和路径
echo "=== 验证 Gurobi 安装 ==="
gurobi_cl --version
echo "GUROBI_HOME: $GUROBI_HOME"
echo "gurobi_cl 路径: $(which gurobi_cl)"

# ========== 第二部分：安装 MacroEnergy ==========
# 5. 进入 MacroEnergy 目录（根据你的实际路径修改）
cd /scratch/gpfs/JENKINS/rl8728/MacroEnergy.jl

# 6. 安装 MacroEnergy 及其所有核心依赖
#    这一步会安装 MacroEnergy 和所有必需的包（但不包括 Gurobi，因为它是可选依赖）
julia --project=. -e 'using Pkg; Pkg.instantiate(); Pkg.precompile()'

# 7. 验证 MacroEnergy 可以正常加载
julia --project=. -e 'using MacroEnergy; println("MacroEnergy loaded successfully")'

# ========== 第三部分：安装和配置 Gurobi ==========
# 8. 检查当前 Gurobi.jl 状态（如果已安装）
echo "=== 检查当前 Gurobi.jl 状态 ==="
julia --project=. -e 'using Pkg; try; pkg_info = Pkg.dependencies()["Gurobi"]; println("Gurobi.jl is installed"); catch; println("Gurobi.jl not installed yet"); end'

# 9. 如果 Gurobi.jl 已安装，先删除它以确保重新构建
#     这样可以确保链接到正确的 Gurobi 12.0.0 版本
echo "=== 清理旧的 Gurobi.jl 安装（如果存在）==="
julia --project=. -e 'using Pkg; try; Pkg.rm("Gurobi"); println("Removed old Gurobi.jl"); catch e; println("Gurobi not in environment or already removed"); end'

# 10. 添加 Gurobi 包（因为 Gurobi 是可选依赖，需要单独添加）
julia --project=. -e 'using Pkg; Pkg.add("Gurobi")'

# 11. 强制 Gurobi.jl 使用系统安装的 Gurobi（而不是下载的版本）
#     删除可能已下载的 Gurobi artifact，强制使用系统安装
echo "=== 清理已下载的 Gurobi artifact（如果存在）==="
julia --project=. -e '
using Pkg
try
    # 尝试删除 Gurobi artifact
    artifacts_toml = joinpath(Pkg.artifacts_toml(), "Artifacts.toml")
    if isfile(artifacts_toml)
        println("Artifacts.toml exists, but we cannot modify it directly")
        println("Gurobi.jl will check GUROBI_HOME first if set correctly")
    end
catch e
    println("Note: ", e)
end
'

# 12. 重建 Gurobi.jl（关键步骤！确保使用 Gurobi 12.0.0）
#     这一步会链接到系统安装的 Gurobi 12.0.0
#     确保环境变量已正确设置，并且设置变量强制使用系统安装
echo "=== 重建 Gurobi.jl（链接到 Gurobi 12.0.0）==="
echo "确保 GUROBI_HOME 指向 Gurobi 12.0.0: $GUROBI_HOME"
echo "检查系统 Gurobi 库: ls -la $GUROBI_HOME/lib/libgurobi*.so"
ls -la $GUROBI_HOME/lib/libgurobi*.so 2>/dev/null || echo "警告: 未找到系统 Gurobi 库文件"

# 在构建时设置环境变量，确保使用系统安装
GUROBI_HOME=$GUROBI_HOME LD_LIBRARY_PATH=$LD_LIBRARY_PATH julia --project=. -e '
using Pkg
# 设置环境变量确保构建时使用系统安装
ENV["GUROBI_HOME"] = get(ENV, "GUROBI_HOME", "")
ENV["LD_LIBRARY_PATH"] = get(ENV, "LD_LIBRARY_PATH", "")
println("Building with GUROBI_HOME: ", ENV["GUROBI_HOME"])
Pkg.build("Gurobi"; verbose=true)
'

# 13. 诊断：检查 Gurobi.jl 实际使用的库路径
#     这可以帮助确认是否链接到了正确的版本
echo "=== 诊断：检查 Gurobi.jl 链接的库 ==="
OMP_NUM_THREADS=1 julia --project=. -e '
using Gurobi
try
    # 检查环境变量
    println("GUROBI_HOME from env: ", get(ENV, "GUROBI_HOME", "not set"))
    # 尝试获取库信息（如果 API 可用）
    if isdefined(Gurobi, :libgurobi)
        lib_path = Gurobi.libgurobi
        println("Gurobi library path: ", lib_path)
        if isfile(lib_path)
            println("✓ Library file exists")
            # 检查是否指向 GUROBI_HOME
            gurobi_home = get(ENV, "GUROBI_HOME", "")
            if gurobi_home != "" && occursin(gurobi_home, lib_path)
                println("✓ Library is in GUROBI_HOME")
            else
                println("⚠ Warning: Library may not be in GUROBI_HOME")
            end
        end
    else
        println("Note: Cannot directly check library path via Gurobi.jl API")
        println("Please verify by checking: ls -la \$GUROBI_HOME/lib/libgurobi*.so")
    end
catch e
    println("Diagnostic info: ", e)
end
'

# 14. 如果诊断显示仍在使用 Gurobi 13，需要运行时强制使用系统库
#     Gurobi.jl 在运行时也会检查 GUROBI_HOME，可以在使用前设置
echo "=== 解决方案：运行时强制使用系统 Gurobi 12.0.0 ==="
echo ""
echo "方法 1: 在 Julia 代码中设置环境变量（推荐）"
echo "在使用 Gurobi 之前，在 Julia 代码中添加："
echo ""
echo "  ENV[\"GUROBI_HOME\"] = \"/usr/licensed/gurobi/12.0.0/linux64\""
echo "  ENV[\"LD_LIBRARY_PATH\"] = ENV[\"GUROBI_HOME\"] * \"/lib:\" * get(ENV, \"LD_LIBRARY_PATH\", \"\")"
echo "  using Gurobi"
echo ""
echo "方法 2: 在运行 Julia 时设置环境变量"
echo "  GUROBI_HOME=$GUROBI_HOME LD_LIBRARY_PATH=\$GUROBI_HOME/lib:\$LD_LIBRARY_PATH julia --project=. your_script.jl"
echo ""
echo "方法 3: 创建启动脚本"
cat << 'EOF' > /tmp/test_gurobi12.jl
# 强制使用系统安装的 Gurobi 12.0.0
ENV["GUROBI_HOME"] = "/usr/licensed/gurobi/12.0.0/linux64"
ENV["LD_LIBRARY_PATH"] = ENV["GUROBI_HOME"] * "/lib:" * get(ENV, "LD_LIBRARY_PATH", "")

using Gurobi
println("Testing Gurobi with system installation...")
try
    # 不创建环境，只测试库加载
    println("✓ Gurobi.jl loaded")
    println("GUROBI_HOME: ", ENV["GUROBI_HOME"])
    if isdefined(Gurobi, :libgurobi)
        lib_path = Gurobi.libgurobi
        println("Library path: ", lib_path)
        if occursin("gurobi120", lib_path) || occursin(ENV["GUROBI_HOME"], lib_path)
            println("✓ Using Gurobi 12.0.0 from system installation")
        else
            println("⚠ Warning: May still be using downloaded version")
            println("  Try setting ENV[\"GUROBI_HOME\"] before 'using Gurobi' in your code")
        end
    end
catch e
    println("Error: ", e)
end
EOF
echo "测试脚本已创建: /tmp/test_gurobi12.jl"
echo "运行测试: GUROBI_HOME=$GUROBI_HOME LD_LIBRARY_PATH=\$GUROBI_HOME/lib:\$LD_LIBRARY_PATH julia --project=. /tmp/test_gurobi12.jl"

# 15. 验证 Gurobi 安装（使用运行时环境变量强制系统库）
#     只验证包可以加载，不测试许可证
echo "=== 验证 Gurobi.jl 可以加载（使用系统 Gurobi 12.0.0）==="
GUROBI_HOME=$GUROBI_HOME LD_LIBRARY_PATH=$GUROBI_HOME/lib:$LD_LIBRARY_PATH OMP_NUM_THREADS=1 julia --project=. -e '
ENV["GUROBI_HOME"] = get(ENV, "GUROBI_HOME", "")
ENV["LD_LIBRARY_PATH"] = ENV["GUROBI_HOME"] * "/lib:" * get(ENV, "LD_LIBRARY_PATH", "")
using Gurobi
println("✓ Gurobi.jl loaded successfully")
println("GUROBI_HOME: ", ENV["GUROBI_HOME"])
if isdefined(Gurobi, :libgurobi)
    lib_path = Gurobi.libgurobi
    println("Library: ", lib_path)
    if occursin(ENV["GUROBI_HOME"], lib_path)
        println("✓ Using system Gurobi from GUROBI_HOME")
    else
        println("⚠ Note: Library path does not match GUROBI_HOME")
        println("  You may need to set ENV[\"GUROBI_HOME\"] before 'using Gurobi' in your code")
    end
end
println("Note: License validation will happen when you first use Gurobi.Optimizer")
'

# 16. 重要提示：如果仍然遇到版本 13 的错误
echo ""
echo "=== 故障排除提示 ==="
echo "如果仍然看到 'license not valid for Gurobi version 13' 错误："
echo "1. 确认模块加载正确: module list | grep gurobi"
echo "2. 确认 GUROBI_HOME 指向 12.0.0: echo \$GUROBI_HOME"
echo "3. 检查库文件: ls -la \$GUROBI_HOME/lib/libgurobi*.so"
echo "4. 可能需要完全清理 Julia 包缓存并重新安装"
echo "5. 联系系统管理员确认许可证是否支持 Gurobi 12.0.0"
